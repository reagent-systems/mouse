http://h/a/b + c -> http://h/a/c
http://h/a/b + /c -> http://h/c
http://h/a/b + ../c -> http://h/c
http://h/a/b + ?q=1 -> http://h/a/b?q=1
http://h/a/b + #f -> http://h/a/b#f
http://h/a/b + //other/x -> http://other/x
http://h/a/b + https://z/y -> https://z/y
http://h/a/ + . -> http://h/a/
http://h:8080/a + b -> http://h:8080/b
/base/path + sub -> /base/sub
